# Calls in Media Center's assemblies that are rewritten at patch time.
#   <declaring type>::<method>  =>  <stand-in method in WmcStandins.Compat>
#   <declaring type>::<method>  =>  @guard        (at each call: an exception goes to
#                                                  WmcStandins.Compat.Swallowed instead of onwards)
#   <declaring type>::<method>  =>  @body <name>  (replace the method's own body by a call to
#                                                  WmcStandins.Compat.<name>; every overload)
#   <declaring type>::<method>  =>  @path         (at each call: the first string argument is a file
#                                                  path; pass it through WmcStandins.Compat.FixPath)
#   <declaring type>::<method>  =>  @path2        (the same for the first two string arguments)
#   <declaring type>::<method>  =>  @default      (at each call: do nothing, return null/0/false)
#   <declaring type>::<method>  =>  @stub         (replace the method's own body the same way;
#                                                  needed for virtual and reflection calls)
# Blank lines and lines starting with # are ignored.

System.Diagnostics.Process::set_PriorityClass => Process_set_PriorityClass
